# Databricks notebook source
import json
import pandas as pd
import numpy as np
import io

# COMMAND ----------

from scipy import stats
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import cross_val_score

# COMMAND ----------

csv_path = "{vol_path}/training_data.csv"
pdf = pd.read_csv(csv_path)
print(f"Shape: {pdf.shape}")

# COMMAND ----------

features = ["f1", "f2", "f3", "f4", "f5", "f6"]
X = pdf[features]
y = pdf["label"]

# Point-biserial correlations
correlations = {}
for feat in features:
    r, p = stats.pointbiserialr(y, X[feat])
    correlations[feat] = abs(r)
    print(f"|corr({feat}, label)| = {abs(r):.4f}  (p={p:.2e})")

# COMMAND ----------

# Single-feature CV accuracy
single_accs = {}
for feat in features:
    Xf = X[[feat]].values
    dt = DecisionTreeClassifier(max_depth=5, random_state=42)
    scores = cross_val_score(dt, Xf, y, cv=5, scoring="accuracy")
    acc = scores.mean()
    single_accs[feat] = float(acc)
    print(f"{feat} CV accuracy: {acc:.4f} +/- {scores.std():.4f}")

# COMMAND ----------

# Training accuracy per feature
train_accs = {}
for feat in features:
    dt_f = DecisionTreeClassifier(max_depth=5, random_state=42)
    dt_f.fit(X[[feat]], y)
    acc = dt_f.score(X[[feat]], y)
    train_accs[feat] = float(acc)
    print(f"{feat} train accuracy: {acc:.4f}")

# All-feature model
dt_all = DecisionTreeClassifier(max_depth=5, random_state=42)
dt_all.fit(X, y)
feat_imp = {features[i]: float(dt_all.feature_importances_[i]) for i in range(len(features))}
print("Feature importances:", feat_imp)

# COMMAND ----------

best_cv = max(single_accs, key=single_accs.get)
best_corr = max(correlations, key=correlations.get)
best_imp = max(feat_imp, key=feat_imp.get)
best_train = max(train_accs, key=train_accs.get)

result = {
    "correlations": correlations,
    "single_feature_cv_accuracy": single_accs,
    "single_feature_train_accuracy": train_accs,
    "feature_importance": feat_imp,
    "best_cv_acc_feat": best_cv,
    "best_corr_feat": best_corr,
    "best_imp_feat": best_imp,
    "best_train_feat": best_train
}

print("RESULT:", json.dumps(result))
dbutils.notebook.exit(json.dumps(result))
