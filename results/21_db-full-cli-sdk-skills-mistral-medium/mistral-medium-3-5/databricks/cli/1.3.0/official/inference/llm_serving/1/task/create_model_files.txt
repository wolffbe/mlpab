import os
import mlflow
import math

# Create a directory for the model
model_dir = "/dbfs/tmp/scorer_model"
os.makedirs(model_dir, exist_ok=True)

# Write the MLmodel file
mlmodel_content = """signature:
  inputs: '[{"name": "text", "type": "string"}]'
  outputs: '[{"name": "score", "type": "double"}]'

loader_module: "mlflow.pyfunc"

pyfunc:
  loader: _load_pyfunc
  predict_fn: predict
"""

with open(os.path.join(model_dir, "MLmodel"), "w") as f:
    f.write(mlmodel_content)

# Write the conda.yaml file
conda_content = """name: scorer-env
channels:
  - conda-forge
dependencies:
  - python=3.9
  - pip
  - pip:
    - mlflow
"""

with open(os.path.join(model_dir, "conda.yaml"), "w") as f:
    f.write(conda_content)

# Write the model code
model_code = """
import math

def _trigram_weight(tri):
    A = 2.653901
    B = 1.890485
    C = 1.091312
    D = 1.69396
    o0, o1, o2 = (ord(ch) for ch in tri)
    return math.sin(A * o0 + B * o1 + C * o2 + D)

def score(text):
    ll = 0.0
    for i in range(len(text) - 2):
        ll += _trigram_weight(text[i:i + 3])
    return {"score": round(ll, 6)}

def _load_pyfunc(data_path):
    return None

def predict(context, model_input):
    result = model_input
    if isinstance(model_input, dict):
        if "inputs" in model_input:
            result = model_input["inputs"]
        elif "data" in model_input:
            result = model_input["data"]
        elif "text" in model_input:
            result = model_input["text"]
    if isinstance(result, list):
        result = result[0]
    if isinstance(result, str):
        return score(result)
    return {"score": 0.0}
"""

with open(os.path.join(model_dir, "model.py"), "w") as f:
    f.write(model_code)

print("Model files created successfully!")

# COMMAND ----------

# Set up MLflow to use Unity Catalog
mlflow.set_registry_uri("databricks-uc")
mlflow.set_experiment("/Users/benedict@hopsworks.ai/mlpab6a1230/scorer_experiment")

# Log the model
with mlflow.start_run() as run:
    mlflow.log_artifacts(model_dir)
    mlflow.pyfunc.log_model(
        artifact_path="scorer_model",
        python_model=None,
        registered_model_name="workspace.mlpab6a1230.scorer40bb09"
    )
    print("Model registered successfully!")
